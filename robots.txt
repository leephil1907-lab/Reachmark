# Reachmark crawler policy
# Private application pages remain crawlable only so search engines can read
# their existing X-Robots-Tag: noindex, nofollow response header.
# Do not disallow /admin, /dashboard or /workspace here: Google must be able
# to crawl them to discover the noindex directive and remove indexed URLs.

User-agent: *
Disallow: /api/admin/
Disallow: /api/internal/
Disallow: /api/private/
Disallow: /data/
Disallow: /backups/
Disallow: /exports/
Disallow: /uploads/private/
Disallow: /*.sqlite3$
Disallow: /*.db$
Disallow: /*.log$
